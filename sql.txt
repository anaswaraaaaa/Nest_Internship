CREATE DATABASE IF NOT EXISTS sfotechnologies_db;
USE sfotechnologies_db;

-- =========================================================
-- 1. OPERATOR AUTHENTICATION ROLES TABLE
-- =========================================================
DROP TABLE IF EXISTS users;
CREATE TABLE users (
    user_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    email VARCHAR(150) UNIQUE NOT NULL,
    password VARCHAR(255) NOT NULL,
    role ENUM('admin', 'qc', 'engineer', 'qa') NOT NULL,
    is_disabled BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- =========================================================
-- 2. MASTER PRODUCT CATALOG BLUEPRINTS TABLE
-- =========================================================
DROP TABLE IF EXISTS products;
CREATE TABLE products (
    product_id INT AUTO_INCREMENT PRIMARY KEY,
    model_no VARCHAR(50) UNIQUE NOT NULL,
    description TEXT NOT NULL,
    image_url LONGTEXT NULL, -- Upgraded to LONGTEXT cleanly to capture local base64 graphics data strings
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

-- =========================================================
-- 3. WORK ORDER BATCH HEADERS TABLE (Tracks parent manifest state)
-- =========================================================
DROP TABLE IF EXISTS work_orders;
CREATE TABLE work_orders (
    order_no VARCHAR(50) PRIMARY KEY,
    initiator_email VARCHAR(150) NOT NULL,
    status ENUM('Open', 'Closed') DEFAULT 'Open',
    create_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    closed_time VARCHAR(50) DEFAULT '—',
    qa_approved_by VARCHAR(150) DEFAULT '—',
    CONSTRAINT fk_order_initiator FOREIGN KEY (initiator_email) REFERENCES users(email) ON UPDATE CASCADE
);

DROP TABLE IF EXISTS order_items;
CREATE TABLE order_items (
    item_uid INT AUTO_INCREMENT PRIMARY KEY,
    order_no VARCHAR(50) NOT NULL,
    model_no VARCHAR(50) NOT NULL,          -- Work Order Number / Model Reference
    description TEXT NOT NULL,              -- Description
    quantity INT NOT NULL,                  -- Quantity
    part_arrangement ENUM('Assembled', 'Loosed') NOT NULL DEFAULT 'Assembled', -- MISSING FIELD FIXED HERE
    selected_checkpoints TEXT NULL,         -- Checklist
    qc_status ENUM('Verified', 'Rejected') DEFAULT 'Verified', -- Status
    remarks TEXT NULL,                      -- Remarks
    image_src LONGTEXT NULL,                -- Reference Image
    CONSTRAINT fk_item_parent_order FOREIGN KEY (order_no) REFERENCES work_orders(order_no) ON DELETE CASCADE ON UPDATE CASCADE,
    CONSTRAINT fk_item_product_ref FOREIGN KEY (model_no) REFERENCES products(model_no) ON UPDATE CASCADE
);
